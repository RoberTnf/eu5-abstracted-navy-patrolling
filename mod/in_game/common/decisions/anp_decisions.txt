# Abstracted Navy Patrolling: patrol every sea zone bordering one of our locations.
anp_patrol_coastal_waters = {
	decision_category = government_decisions
	hide_repeatable_tag = yes

	potential = {
		is_ai = no
		num_of_ports > 0
	}

	allow = {
		always = yes
	}

	ai_will_do = {
		value = 0
	}

	option = {
		name = anp_patrol_coastal_waters.a
		effect = {
			every_owned_location = {
				limit = { is_coastal = yes }
				every_neighbor_location = {
					limit = { anp_is_patrollable_zone = yes }
					save_temporary_scope_as = anp_new_zone
					root = {
						if = {
							limit = {
								NOT = { is_target_in_variable_list = { name = anp_zones target = scope:anp_new_zone } }
							}
							add_to_variable_list = { name = anp_zones target = scope:anp_new_zone }
						}
					}
				}
			}
		}
	}
}

# Abstracted Navy Patrolling: stop patrolling every sea zone (fleets fall back to their own sea area).
anp_stop_all_patrols = {
	decision_category = government_decisions
	hide_repeatable_tag = yes

	potential = {
		is_ai = no
		has_variable_list = anp_zones
	}

	allow = {
		always = yes
	}

	ai_will_do = {
		value = 0
	}

	option = {
		name = anp_stop_all_patrols.a
		effect = {
			every_in_list = {
				variable = anp_zones
				remove_location_modifier = anp_patrolled_waters
			}
			clear_variable_list = anp_zones
		}
	}
}
